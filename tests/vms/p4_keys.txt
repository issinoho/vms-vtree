# PTYDRIVE script for Phase 4, inside the work directory's [.LIVE] (test_live.com P1).
# Positioned with filespecs; no printing (it would reach the user's real queue).
2 dcl | set terminal/width=80/page=24\r
5 start | vtree [.live]\r
2 files | \r
2 only notes;2 | f\x15NOTES.TXT;2\r
3 attributes | a
2 prot prompt | p
2 prot | \x15G:R,W:R\r
3 prot yes | y
2 owner prompt | o
2 owner self | [200,101]\r
3 owner yes | y
2 ace prompt | a
2 ace | \x15(IDENTIFIER=[*,*],ACCESS=READ)\r
3 ace yes | y
2 expires prompt | e
2 expires | \x151-JAN-2030\r
3 expires yes | y
2 nobackup | b
3 nobackup yes | y
2 erase | z
3 erase yes | y
3 after changes | \x20
2 del ace | \e[3~
3 del ace yes | y
3 after ace delete | \x20
2 back | \e[21~
2 both notes | f\x15NOTES.TXT\r
2 tag both | \eOPt
2 search prompt | s
4 search | \x15line 4\r
2 search back | \x20
4 compare | \eOPv
2 compare back | \e[21~
2 quit | \e[21~\e[21~
5 privileged start | vtree/privileges [.live]\r
3 privileged quit | \e[21~
